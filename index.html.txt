function getPeriodForMonth(year, month, startDay) {
  const day = clampStartDay(startDay == null ? state.budgetStartDay : startDay);
  const start = new Date(year, month, day);
  const end = new Date(year, month + 1, day - 1);
  return { start: toDateString(start), end: toDateString(end), year: start.getFullYear(), month: start.getMonth() };
}
function getBudgetPeriod(referenceDate, offset, startDay) {
  const day = clampStartDay(startDay == null ? state.budgetStartDay : startDay);
  const ref = referenceDate instanceof Date ? referenceDate : parseDate(referenceDate || todayString());
  let month = ref.getMonth();
  if (ref.getDate() < day) month -= 1;
  return getPeriodForMonth(ref.getFullYear(), month + (offset || 0), day);
}

}

function clampDate(year, month, day) {
  const lastDay = new Date(year, month + 1, 0).getDate();
  return new Date(year, month, Math.min(day, lastDay));
}
function listOccurrences(r, untilString) {
  const result = [];
  if (!r || !r.start_date) return result;
  const start = parseDate(r.start_date);
  const endDate = r.end_date ? String(r.end_date).slice(0, 10) : null;
  const limit = endDate && endDate < untilString ? endDate : untilString;
  for (let i = 0; i < 1000; i++) {
    let date;
    if (r.frequency === "weekly") date = new Date(start.getFullYear(), start.getMonth(), start.getDate() + 7 * i);
    else if (r.frequency === "yearly") date = clampDate(start.getFullYear() + i, start.getMonth(), start.getDate());
    else date = clampDate(start.getFullYear(), start.getMonth() + i, start.getDate());
    const text = toDateString(date);
    if (text > limit) break;
    result.push(text);
  }
  return result;
}
function lastGeneratedDate(recurringId) {
  let last = null;
  state.transactions.forEach(function (t) {
    if (sameId(t.recurring_id, recurringId)) {
      const d = String(t.transaction_date).slice(0, 10);
      if (!last || d > last) last = d;
    }
  });
  return last;
}
function pendingOccurrences(r, untilString) {
  const last = lastGeneratedDate(r.id);
  return listOccurrences(r, untilString).filter(function (d) { return !last || d > last; });
}
function nextOccurrenceAfter(r, dateString) {
  return listOccurrences(r, addDays(dateString, 800)).find(function (d) { return d > dateString; }) || null;
}
function firstOccurrenceOnOrAfter(r, dateString) {
  return listOccurrences(r, addDays(dateString, 800)).find(function (d) { return d >= dateString; }) || null;
}
function monthlyEquivalent(r) {
  const amount = Number(r.amount) || 0;
  if (r.frequency === "weekly") return amount * 52 / 12;
  if (r.frequency === "yearly") return amount / 12;
  return amount;
}
function adjustForPause(recurring, update) {
  const today = todayString();
  const missed = pendingOccurrences(recurring, today).filter(function (d) { return d < today; });
  if (!missed.length) return;
  const ok = confirm("Ce paiement récurrent était en pause.\n\nOK : ajouter les " + missed.length + " échéance(s) passée(s) à l'historique.\nAnnuler : reprendre seulement à partir d'aujourd'hui.");
  if (!ok) {
    const next = firstOccurrenceOnOrAfter(recurring, today);
    if (next) update.start_date = next;
  }
}
async function insertRowsOneByOne(rows) {
  for (const row of rows) {
    const result = await state.client.from("transactions").insert(row);
    if (result.error && result.error.code !== "23505") throw result.error;
  }
}
async function generateRecurringTransactions() {
  const today = todayString();
  const rows = [];
  state.recurring.forEach(function (r) {
    if (!r.active || !r.start_date) return;
    pendingOccurrences(r, today).forEach(function (d) {
      rows.push({
        user_id: state.user.id, type: r.type, amount: r.amount,
        category_id: r.category_id == null ? null : r.category_id,
        transaction_date: d, note: r.name, recurring_id: r.id,
        occurrence_key: r.id + "_" + d
      });
    });
  });
  for (let i = 0; i < rows.length; i += 500) {
    const chunk = rows.slice(i, i + 500);
    const result = await state.client.from("transactions").upsert(chunk, { onConflict: "user_id,occurrence_key", ignoreDuplicates: true });
    if (result.error) {
      if (result.error.code === "42P10") await insertRowsOneByOne(chunk);
      else throw result.error;
    }
  }
  return rows.length;
}

async function fetchAllRows(table, build) {
  const pageSize = 1000;
  let from = 0;
  let rows = [];
  while (true) {
    const query = build(state.client.from(table).select("*")).range(from, from + pageSize - 1);
    const result = await query;
    if (result.error) throw result.error;
    const data = result.data || [];
    rows = rows.concat(data);
    if (data.length < pageSize) break;
    from += pageSize;
  }
  return rows;
}
async function loadAllData() {
  const uid = state.user.id;
  const results = await Promise.all([
    fetchAllRows("categories", function (q) { return q.eq("user_id", uid).order("name", { ascending: true }).order("id", { ascending: true }); }),
    fetchAllRows("recurring_payments", function (q) { return q.eq("user_id", uid).order("created_at", { ascending: true }).order("id", { ascending: true }); }),
    fetchAllRows("transactions", function (q) { return q.eq("user_id", uid).order("transaction_date", { ascending: false }).order("created_at", { ascending: false }).order("id", { ascending: false }); })
  ]);
  state.categories = results[0];
  state.recurring = results[1];
  state.transactions = results[2];
  state.categoryById = new Map(state.categories.map(function (c) { return [String(c.id), c]; }));
}

async function refreshData(options) {
  const opts = options || {};
  if (!state.client || !state.user) return;
  if (state.loadingData) { state.pendingRefresh = true; return; }
  state.loadingData = true;
  setSyncStatus("busy");
  try {
    await loadAllData();
    if (opts.generateRecurring !== false) {
      const added = await generateRecurringTransactions();
      if (added > 0) await loadAllData();
    }
    renderAll();
    setSyncStatus("ok");
    if (state.syncErrorShown) { hideMessage("appMessage"); state.syncErrorShown = false; }
  } catch (error) {
    console.error(error);
    setSyncStatus("error");
    showMessage("appMessage", "Synchronisation impossible : " + explainError(error), "error");
    state.syncErrorShown = true;
  } finally {
    state.loadingData = false;
    if (state.pendingRefresh) { state.pendingRefresh = false; scheduleRefresh(300); }
  }
}
function scheduleRefresh(delay) {
  clearTimeout(state.refreshTimer);
  state.refreshTimer = setTimeout(function () { refreshData(); }, delay == null ? 500 : delay);
}
function subscribeRealtime() {
  unsubscribeRealtime();
  if (!state.client || !state.user) return;
  const filter = "user_id=eq." + state.user.id;
  const onChange = function () { scheduleRefresh(); };
  try {
    state.channel = state.client.channel("budget-" + state.user.id)
      .on("postgres_changes", { event: "*", schema: "public", table: "transactions", filter: filter }, onChange)
      .on("postgres_changes", { event: "*", schema: "public", table: "categories", filter: filter }, onChange)
      .on("postgres_changes", { event: "*", schema: "public", table: "recurring_payments", filter: filter }, onChange)
      .subscribe();
  } catch (error) { console.error(error); }
}
function unsubscribeRealtime() {
  if (state.channel && state.client) { try { state.client.removeChannel(state.channel); } catch (e) { console.error(e); } }
  state.channel = null;
}

function totalBalanceUntil(limitDate) {
  return state.transactions.reduce(function (sum, t) {
    if (String(t.transaction_date).slice(0, 10) > limitDate) return sum;
    const amount = Number(t.amount) || 0;
    return t.type === "income" ? sum + amount : sum - amount;
  }, 0);
}

function renderUpcoming() {
  const today = todayString();
  const limit = addDays(today, 30);
  const items = [];
  state.recurring.forEach(function (r) {
    if (!r.active) return;
    listOccurrences(r, limit).forEach(function (d) { if (d > today) items.push({ r: r, d: d }); });
  });
  items.sort(function (a, b) { return a.d < b.d ? -1 : a.d > b.d ? 1 : 0; });
  $("homeUpcoming").innerHTML = items.length ? items.slice(0, 6).map(function (item) {
    const c = getCategory(item.r.category_id);
    const income = item.r.type === "income";
    return '<div class="item"><span class="dot" style="background:' + safeColor(c.color) + '"></span><div class="item-main"><strong>' + escapeHtml(item.r.name) + '</strong><small>' + escapeHtml(c.name) + " · " + formatDate(item.d, { weekday: "short", day: "numeric", month: "short" }) + '</small></div><div class="item-amount ' + (income ? "green" : "red") + '">' + (income ? "+" : "−") + formatMoney(item.r.amount) + "</div></div>";
  }).join("") : emptyHtml("Aucune échéance prévue dans les 30 prochains jours.");
}
function renderHistory() {
  const showAll = $("historyAll").checked;
  const period = getBudgetPeriod(new Date(), state.historyOffset);
  const label = formatBudgetPeriod(period);
  $("historyPeriodTitle").textContent = label.title;
  $("historyPeriodRange").textContent = label.range;
  $("historyNav").classList.toggle("disabled", showAll);
  const type = $("historyType").value;
  const category = $("historyCategory").value || "all";
  const search = $("historySearch").value.trim().toLowerCase();
  const list = state.transactions.filter(function (t) {
    if (!showAll && !isDateInsideBudgetPeriod(t.transaction_date, period)) return false;
    if (type !== "all" && t.type !== type) return false;
    if (category === "none" && t.category_id != null) return false;
    if (category !== "all" && category !== "none" && !sameId(t.category_id, category)) return false;
    if (search) {
      const text = ((t.note || "") + " " + getCategory(t.category_id).name).toLowerCase();
      if (text.indexOf(search) === -1) return false;
    }
    return true;
  });
  const income = sumAmounts(list.filter(function (t) { return t.type === "income"; }));
  const expense = sumAmounts(list.filter(function (t) { return t.type !== "income"; }));
  $("historySummary").textContent = list.length + " opération" + (list.length > 1 ? "s" : "") + " · revenus " + formatMoney(income) + " · dépenses " + formatMoney(expense);
  if (!list.length) { $("historyList").innerHTML = emptyHtml("Aucune opération ne correspond."); return; }
  let html = list.slice(0, MAX_HISTORY_ROWS).map(function (t) { return transactionRowHtml(t, true); }).join("");
  if (list.length > MAX_HISTORY_ROWS) html += emptyHtml("… et " + (list.length - MAX_HISTORY_ROWS) + " autres opérations. Utilisez les filtres pour les afficher.");
  $("historyList").innerHTML = html;
}

function makeChart(key, canvasId, config) {
  if (state.charts[key]) { try { state.charts[key].destroy(); } catch (e) {} delete state.charts[key]; }
  const canvas = $(canvasId);
  if (!canvas || typeof Chart === "undefined") return;
  state.charts[key] = new Chart(canvas, config);
}
function destroyCharts() {
  Object.keys(state.charts).forEach(function (k) { try { state.charts[k].destroy(); } catch (e) {} });
  state.charts = {};
}
function chartOptions(settings) {
  const s = settings || {};
  return {
    responsive: true, maintainAspectRatio: false,
    interaction: { mode: "index", intersect: false },
    plugins: {
      legend: { position: "bottom", labels: { boxWidth: 12, usePointStyle: true } },
      tooltip: {
        filter: function (item) { return !s.stacked || (item.parsed && item.parsed.y > 0); },
 callbacks: { label: function (ctx) { const v = ctx.parsed && typeof ctx.parsed === "object" ? ctx.parsed.y : ctx.parsed; return " " + (ctx.dataset.label || "") + " : " + formatMoney(v); } }
      }
    },
    scales: {
      x: { stacked: !!s.stacked, grid: { display: false } },
      y: { stacked: !!s.stacked, beginAtZero: s.beginAtZero !== false, ticks: { callback: function (v) { return compactMoney(v); } } }
    }
  };
}
function renderStats() {
  const year = state.statsYear || getBudgetPeriod(new Date()).year;
  const yearTx = state.transactions.filter(function (t) { return budgetMonthOf(t.transaction_date).year === year; });
  const months = [];
  for (let i = 0; i < 12; i++) months.push({ i: i, income: 0, expense: 0, byCat: new Map() });
  yearTx.forEach(function (t) {
    const m = months[budgetMonthOf(t.transaction_date).month];
    const amount = Number(t.amount) || 0;
    if (t.type === "income") { m.income += amount; return; }
    m.expense += amount;
    const key = t.category_id == null ? "none" : String(t.category_id);
    m.byCat.set(key, (m.byCat.get(key) || 0) + amount);
  });
  const income = months.reduce(function (s, m) { return s + m.income; }, 0);
  const expense = months.reduce(function (s, m) { return s + m.expense; }, 0);
  const balance = income - expense;
  $("statsIncome").textContent = formatMoney(income);
  $("statsExpense").textContent = formatMoney(expense);
  $("statsBalance").textContent = formatMoney(balance);
  $("statsBalance").className = balance >= 0 ? "green" : "red";
  $("statsRate").textContent = income > 0 ? (balance / income * 100).toFixed(1).replace(".", ",") + " %" : "—";
  let note = state.budgetStartDay === 1
    ? "Mois calendaires (du 1er au dernier jour du mois)."
    : "Mois budgétaires : du " + state.budgetStartDay + " au " + (state.budgetStartDay - 1) + " du mois suivant. Chaque mois porte le nom du mois où il commence.";
  if (typeof Chart === "undefined") note += " ⚠️ Graphiques indisponibles : la bibliothèque Chart.js n'a pas pu être chargée.";
  $("statsPeriodNote").textContent = note;

  const current = getBudgetPeriod(new Date());
  const lastIndex = year < current.year ? 11 : (year > current.year ? -1 : current.month);
  const groups = groupExpenses(yearTx);
  const activeMonths = months.filter(function (m) { return m.income > 0 || m.expense > 0; });
  const labels = MONTH_SHORT.slice();

  makeChart("monthly", "chartMonthly", {
    type: "bar",
    data: { labels: labels, datasets: [
      { label: "Revenus", data: months.map(function (m) { return round2(m.income); }), backgroundColor: "#22c55e", borderRadius: 6 },
      { label: "Dépenses", data: months.map(function (m) { return round2(m.expense); }), backgroundColor: "#ef4444", borderRadius: 6 }
    ] },
    options: chartOptions()
  });
  let running = 0;
  const cumulative = months.map(function (m, i) { running += m.income - m.expense; return i <= lastIndex ? round2(running) : null; });
  makeChart("cumulative", "chartCumulative", {
    type: "line",
    data: { labels: labels, datasets: [{ label: "Solde cumulé", data: cumulative, borderColor: "#4f46e5", backgroundColor: "rgba(79,70,229,0.12)", fill: true, tension: 0.3, pointRadius: 3 }] },
    options: chartOptions({ beginAtZero: false })
  });
  makeChart("categories", "chartCategories", {
    type: "doughnut",
    data: { labels: groups.map(function (g) { return g.name; }), datasets: [{ data: groups.map(function (g) { return round2(g.total); }), backgroundColor: groups.map(function (g) { return safeColor(g.color); }), borderColor: "#fff", borderWidth: 2 }] },
    options: {
      responsive: true, maintainAspectRatio: false, cutout: "58%",
      plugins: {
        legend: { position: "bottom", labels: { boxWidth: 12, usePointStyle: true } },
        tooltip: { callbacks: { label: function (ctx) {
   const total = ctx.dataset.data.reduce(function (a, b) { return a + b; }, 0);
          return " " + ctx.label + " : " + formatMoney(ctx.parsed) + " (" + (total ? Math.round(ctx.parsed / total * 100) : 0) + " %)";
 } } }
      }
    }
  });
  const top = groups.slice(0, 7);
  const others = groups.slice(7);
  const stackedSets = top.map(function (g) {
    return { label: g.name, data: months.map(function (m) { return round2(m.byCat.get(g.key) || 0); }), backgroundColor: safeColor(g.color) };
  });
  if (others.length) stackedSets.push({ label: "Autres", data: months.map(function (m) { return round2(others.reduce(function (s, g) { return s + (m.byCat.get(g.key) || 0); }, 0)); }), backgroundColor: "#cbd5e1" });
  makeChart("stacked", "chartStacked", { type: "bar", data: { labels: labels, datasets: stackedSets }, options: chartOptions({ stacked: true }) });

  renderStatsInsights(year, activeMonths, groups, income, expense);
  renderStatsRanking(groups, expense, activeMonths.length);
}
function renderStatsInsights(year, activeMonths, groups, income, expense) {
  const box = $("statsInsights");
  if (!activeMonths.length) { box.innerHTML = emptyHtml("Aucune opération pour " + year + "."); return; }
  const lines = [];
  const best = activeMonths.reduce(function (a, b) { return (b.income - b.expense) > (a.income - a.expense) ? b : a; });
  const worst = activeMonths.reduce(function (a, b) { return b.expense > a.expense ? b : a; });
  lines.push("📈 Meilleur mois : <strong>" + MONTH_NAMES[best.i] + "</strong>, avec un solde de " + formatMoney(best.income - best.expense) + ".");
  if (worst.expense > 0) lines.push("💸 Mois le plus dépensier : <strong>" + MONTH_NAMES[worst.i] + "</strong> (" + formatMoney(worst.expense) + ").");
  if (groups.length && expense > 0) lines.push("🏷️ Premier poste de dépenses : <strong>" + escapeHtml(groups[0].name) + "</strong>, soit " + Math.round(groups[0].total / expense * 100) + " % des dépenses.");
  lines.push("📅 Dépense moyenne : <strong>" + formatMoney(expense / activeMonths.length) + "</strong> par mois (sur " + activeMonths.length + " mois avec des opérations).");
  if (income > 0) {
    const rate = (income - expense) / income * 100;
    const comment = rate >= 20 ? "excellent, continuez ainsi ! 🎉" : rate >= 10 ? "correct, vous mettez de côté régulièrement." : rate >= 0 ? "faible : essayez de réduire un poste de dépenses." : "négatif : vous avez dépensé plus que vos revenus.";
    lines.push("💰 Taux d'épargne de " + rate.toFixed(1).replace(".", ",") + " % : " + comment);
  }
  const fixed = state.recurring.filter(function (r) { return r.active && r.type !== "income"; }).reduce(function (s, r) { return s + monthlyEquivalent(r); }, 0);
  const avgIncome = income / activeMonths.length;
  if (fixed > 0) lines.push("🔁 Vos charges récurrentes actives représentent environ <strong>" + formatMoney(fixed) + "</strong> par mois" + (avgIncome > 0 ? ", soit " + Math.round(fixed / avgIncome * 100) + " % de vos revenus mensuels moyens" : "") + ".");
  box.innerHTML = lines.map(function (l) { return '<div class="insight">' + l + "</div>"; }).join("");
}
function renderStatsRanking(groups, expense, monthsCount) {
  const box = $("statsRanking");
  if (!groups.length) { box.innerHTML = emptyHtml("Aucune dépense sur cette année."); return; }
  box.innerHTML = groups.map(function (g, index) {
    const pct = expense > 0 ? g.total / expense * 100 : 0;
    const color = safeColor(g.color);
    return '<div class="bar-row"><div class="bar-head"><strong>' + (index + 1) + '.</strong><span class="dot" style="background:' + color + '"></span><span class="name">' + escapeHtml(g.name) + '</span><span class="muted">' + Math.round(pct) + ' %</span><strong>' + formatMoney(g.total) + '</strong></div><div class="bar"><div style="width:' + Math.max(pct, 2) + '%;background:' + color + '"></div></div><small class="muted">≈ ' + formatMoney(g.total / Math.max(monthsCount, 1)) + " par mois</small></div>";
  }).join("");
}

async function runWithForm(form, messageId, task) {
  if (!state.client) { showMessage(messageId, "Supabase n'est pas configuré : vérifiez SUPABASE_URL et SUPABASE_PUBLIC_KEY.", "error"); return; }
  const button = form ? form.querySelector('button[type="submit"]') : null;
  const label = button ? button.textContent : "";
  if (button) { button.disabled = true; button.textContent = "Patientez…"; }
  try { await task(); }
  catch (error) {
    console.error(error);
    if (messageId === "appMessage") notify(explainError(error), "error");
    else showMessage(messageId, explainError(error), "error");
  } finally {
    if (button) { button.disabled = false; if (button.textContent === "Patientez…") button.textContent = label; }
  }
}
async function handleLogin(event) {
  event.preventDefault();
  const form = event.currentTarget;
  const email = $("loginEmail").value.trim();
  const password = $("loginPassword").value;
  hideMessage("authMessage");
  await runWithForm(form, "authMessage", async function () {
    const result = await state.client.auth.signInWithPassword({ email: email, password: password });
    if (result.error) throw result.error;
    $("loginPassword").value = "";
    await enterApplication(result.data.user);
  });
}
async function handleSignup(event) {
  event.preventDefault();
  const form = event.currentTarget;
  const email = $("signupEmail").value.trim();
  const password = $("signupPassword").value;
  if (password.length < 6) { showMessage("authMessage", "Le mot de passe doit contenir au moins 6 caractères.", "error"); return; }
  if (password !== $("signupConfirm").value) { showMessage("authMessage", "Les deux mots de passe ne sont pas identiques.", "error"); return; }
  await runWithForm(form, "authMessage", async function () {
    const result = await state.client.auth.signUp({ email: email, password: password, options: { emailRedirectTo: appUrl() } });
    if (result.error) throw result.error;
    const user = result.data.user;
    if (user && Array.isArray(user.identities) && user.identities.length === 0) {
      showMessage("authMessage", "Un compte existe déjà avec cette adresse : connectez-vous ou utilisez « Mot de passe oublié ? ».", "error");
      return;
    }
    form.reset();
    if (result.data.session) { await enterApplication(user); return; }
    showAuthForm("loginForm");
    $("loginEmail").value = email;
    showMessage("authMessage", "Compte créé ! Ouvrez l'e-mail de confirmation (vérifiez aussi les courriers indésirables), puis connectez-vous.", "success");
  });
}
async function handleForgotPassword(event) {
  event.preventDefault();
  const form = event.currentTarget;
  const email = $("forgotEmail").value.trim();
  await runWithForm(form, "authMessage", async function () {
    const result = await state.client.auth.resetPasswordForEmail(email, { redirectTo: appUrl() });
    if (result.error) throw result.error;
    showAuthForm("loginForm");
    $("loginEmail").value = email;
    showMessage("authMessage", "Si un compte existe avec cette adresse, un e-mail de réinitialisation vient d'être envoyé. Ouvrez le lien sur cet appareil.", "success");
  });
}
async function handleResetPassword(event) {
  event.preventDefault();
  const form = event.currentTarget;
  const password = $("resetPassword").value;
  if (password.length < 6) { showMessage("authMessage", "Le mot de passe doit contenir au moins 6 caractères.", "error"); return; }
  if (password !== $("resetConfirm").value) { showMessage("authMessage", "Les deux mots de passe ne sont pas identiques.", "error"); return; }
  await runWithForm(form, "authMessage", async function () {
    const result = await state.client.auth.updateUser({ password: password });
    if (result.error) throw result.error;
    state.recoveryMode = false;
    history.replaceState(null, "", location.pathname);
    form.reset();
    state.user = null;
    await enterApplication(result.data.user);
    notify("Mot de passe modifié avec succès.");
  });
}
async function handleLogout() {
  if (!confirm("Voulez-vous vous déconnecter ?")) return;
  try {
    const result = await state.client.auth.signOut();
    if (result && result.error) throw result.error;
  } catch (error) {
    console.error(error);
    try { Object.keys(localStorage).forEach(function (k) { if (k.indexOf("sb-") === 0) localStorage.removeItem(k); }); } catch (e) {}
  }
  showAuth("loginForm");
}

  try {
    const response = await fetch(CLEAN_URL + "/auth/v1/health", { headers: { apikey: CLEAN_KEY }, signal: controller ? controller.signal : undefined });
    if (response.ok) showMessage("authMessage", "✅ Le serveur Supabase répond. Si la connexion échoue encore, vérifiez votre e-mail et votre mot de passe.", "success");
    else if (response.status === 401 || response.status === 403) showMessage("authMessage", "⚠️ Le serveur répond mais refuse la clé (code " + response.status + "). Vérifiez SUPABASE_PUBLIC_KEY.", "error");
    else showMessage("authMessage", "⚠️ Le serveur répond avec le code " + response.status + ". Le projet est peut-être en pause : ouvrez le tableau de bord Supabase et cliquez sur « Resume project ».", "error");
  } catch (error) {
    showMessage("authMessage", "❌ Serveur injoignable (" + (error.name === "AbortError" ? "délai dépassé" : error.message) + ").\nCauses possibles : projet Supabase en pause, adresse SUPABASE_URL incorrecte, bloqueur de contenu, VPN ou Relais privé, Wi-Fi filtré.", "error");
  } finally { clearTimeout(timer); }
}
function handleAuthEvent(event, session) {
  if (event === "PASSWORD_RECOVERY") {
    state.recoveryMode = true;
    unsubscribeRealtime();
    $("appView").classList.add("hidden");
    $("authView").classList.remove("hidden");
    showAuthForm("resetForm");
    setLoading(false);
    return;
  }
  if (state.recoveryMode) return;
  if (event === "SIGNED_OUT") { if (state.user) showAuth("loginForm"); return; }
  if (!session || !session.user) return;
  if (!state.user || state.user.id !== session.user.id) {
    if (event === "SIGNED_IN") enterApplication(session.user);
    return;
  }
  state.user = session.user;
  const day = clampStartDay((session.user.user_metadata || {}).budget_month_start_day);
  if (day !== state.budgetStartDay) { state.budgetStartDay = day; renderAll(); }
}
function readAuthInfoFromUrl() {
  const text = (location.hash || "") + "&" + (location.search || "");
  return { recovery: /type=recovery/.test(text), error: /error_code=|error=access_denied/.test(text) };
}

async function deleteTransaction(id) {
  const t = state.transactions.find(function (x) { return sameId(x.id, id); });
  if (!t) return;
  let text = "Supprimer cette opération ?";
  const r = t.recurring_id ? state.recurring.find(function (x) { return sameId(x.id, t.recurring_id); }) : null;
  if (r && r.active && lastGeneratedDate(r.id) === String(t.transaction_date).slice(0, 10)) {
    text += "\n\nAttention : c'est la dernière échéance du paiement récurrent « " + r.name + " ». Elle sera recréée automatiquement tant qu'il reste actif. Pour l'arrêter, mettez-le en pause ou ajoutez une date de fin.";
  }
  if (!confirm(text)) return;
  try {
    const result = await state.client.from("transactions").delete().eq("id", id);
    if (result.error) throw result.error;
    if (sameId($("transactionId").value, id)) resetTransactionForm();
    notify("Opération supprimée.");
    await refreshData({ generateRecurring: false });
  } catch (error) { console.error(error); notify(explainError(error), "error"); }
}

async function handleRecurringSubmit(event) {
  event.preventDefault();
  const form = event.currentTarget;
  const id = $("recurringId").value;
  const name = $("recurringName").value.trim();
  const amount = parseAmount($("recurringAmount").value);
  const start = $("recurringStart").value;
  const end = $("recurringEnd").value;
  if (!name) { notify("Indiquez un nom.", "error"); return; }
  if (!amount) { notify("Indiquez un montant valide (ex. 450 ou 12,99).", "error"); return; }
  if (!start) { notify("Indiquez une date de début.", "error"); return; }
  if (end && end < start) { notify("La date de fin doit être après la date de début.", "error"); return; }
  const row = { name: name, type: $("recurringType").value, amount: amount, category_id: $("recurringCategory").value || null, frequency: $("recurringFrequency").value, start_date: start, end_date: end || null, active: $("recurringActive").checked };
  const previous = id ? state.recurring.find(function (r) { return sameId(r.id, id); }) : null;
  let rebuild = false;
  if (previous && (String(previous.start_date).slice(0, 10) !== start || previous.frequency !== row.frequency) && lastGeneratedDate(previous.id)) {
    rebuild = confirm("Vous avez changé la date de début ou la fréquence.\n\nOK : supprimer les opérations déjà créées par ce récurrent et les recréer avec les nouvelles dates.\nAnnuler : garder l'historique tel quel (seules les prochaines échéances suivront les nouvelles dates).");
  }
  if (previous && !previous.active && row.active && !rebuild) adjustForPause(Object.assign({}, previous, row), row);
  await runWithForm(form, "appMessage", async function () {
    let result;
    if (previous) {
      result = await state.client.from("recurring_payments").update(row).eq("id", id);
      if (result.error) throw result.error;
      if (rebuild) {
        result = await state.client.from("transactions").delete().eq("recurring_id", id);
        if (result.error) throw result.error;
      }
      if (row.end_date) {
        result = await state.client.from("transactions").delete().eq("recurring_id", id).gt("transaction_date", row.end_date);
        if (result.error) throw result.error;
      }
    } else {
      result = await state.client.from("recurring_payments").insert(Object.assign({}, row, { user_id: state.user.id }));
      if (result.error) throw result.error;
    }
    resetRecurringForm();
    notify(previous ? "Paiement récurrent modifié." : "Paiement récurrent ajouté.");
    await refreshData();
  });
}

}

async function initializeApplication() {
  setLoading(true, "Connexion au serveur…");
  const urlInfo = readAuthInfoFromUrl();
  if (!window.supabase || typeof window.supabase.createClient !== "function") {
    throw new Error("La bibliothèque Supabase n'a pas pu être téléchargée. Vérifiez votre connexion Internet, désactivez les bloqueurs de contenu, puis rechargez la page.");
  }
  const problem = getConfigurationProblem();
  if (problem) {
    showConfigurationWarning(CONFIG_MESSAGES[problem]);
    showAuth("loginForm");
    return;
  }
  state.recoveryMode = urlInfo.recovery;
  state.client = window.supabase.createClient(CLEAN_URL, CLEAN_KEY, {
    auth: { persistSession: true, autoRefreshToken: true, detectSessionInUrl: true }
  });
  state.client.auth.onAuthStateChange(function (event, session) {
    setTimeout(function () { handleAuthEvent(event, session); }, 0);
  });
  if (urlInfo.error) {
    history.replaceState(null, "", location.pathname);
    showAuth("loginForm");
    showMessage("authMessage", "Le lien reçu par e-mail est invalide ou a expiré. Demandez un nouveau lien.", "error");
    return;
  }
  const result = await state.client.auth.getSession();
  if (result.error) throw result.error;
  if (urlInfo.recovery) {
    showAuth("resetForm");
    return;
  }
  if (location.hash && /access_token|refresh_token/.test(location.hash)) history.replaceState(null, "", location.pathname + location.search);
  const session = result.data ? result.data.session : null;
  if (session && session.user) await enterApplication(session.user);
  else showAuth("loginForm");
}

const safetyTimer = setTimeout(function () {
  const screen = $("loadingScreen");
  if (screen && !screen.classList.contains("hidden")) {
    setLoading(false);
    showConfigurationWarning("⏳ Le chargement prend trop de temps. Vérifiez votre connexion Internet et que votre projet Supabase n'est pas en pause (tableau de bord Supabase → « Resume project »), puis rechargez la page.");
    if ($("appView").classList.contains("hidden")) $("authView").classList.remove("hidden");
  }
}, 12000);

try { initializeFields(); bindEvents(); } catch (error) { console.error(error); }

initializeApplication()
  .then(function () { clearTimeout(safetyTimer); })
  .catch(function (error) {
    console.error(error);
    clearTimeout(safetyTimer);
    setLoading(false);
    showConfigurationWarning("❌ Erreur au démarrage : " + escapeHtml(explainError(error)));
    if ($("appView").classList.contains("hidden")) $("authView").classList.remove("hidden");
  });
</script>
